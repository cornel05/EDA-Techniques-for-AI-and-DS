Run this script locally to compute real class counts and pixel statistics, then paste the values above.
import scipy.io as sio
import numpy as np

# Load SVHN .mat file (train_32x32.mat or test_32x32.mat)
mat = sio.loadmat("train_32x32.mat")
X = mat["X"]  # shape: (32, 32, 3, N)
y = mat["y"].reshape(-1)

# Label 10 in SVHN means digit 0
y = np.where(y == 10, 0, y)

# Class balance
classes, counts = np.unique(y, return_counts=True)
print("Class counts:")
for c, n in zip(classes, counts):
    print(f"digit {c}: {n}")

# Pixel stats in [0,1]
Xf = X.astype(np.float32) / 255.0
Xf = np.transpose(Xf, (3, 0, 1, 2))  # (N, H, W, C)
means = Xf.mean(axis=(0, 1, 2))
stds = Xf.std(axis=(0, 1, 2))

print("Means (R,G,B):", means)
print("Stds  (R,G,B):", stds)

# Dimension check
print("Image shape:", X.shape[:3])  # expected (32, 32, 3)
